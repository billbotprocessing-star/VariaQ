        const cleanFormatting = (text) => {

                .replace(/###\s?/g, '') // Remove ###
                .replace(/\*\*/g, '')   // Remove **
                .trim();

        async function handleFileUpload(file) {
            if (!file) return;
            loading = true;
            render();

            const reader = new FileReader();
            reader.onload = async (e) => {
                try {
                    const base64 = e.target.result.split(',')[1];
                    const mediaType = file.type || (file.name.endsWith('.pdf') ? 'application/pdf' : 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');

                    const response = await fetch('https://billbotprocessing.app.n8n.cloud/webhook/financial-parser', {
                        method: 'POST',
                        body: JSON.stringify({ file: base64, mediaType: mediaType, fileName: file.name }),
                        headers: { 'Content-Type': 'application/json' }
                    });

                    const data = await response.json();
                    console.log("Cleaned Data from n8n:", data); // This should now show your raw numbers
                    
                    if (data && typeof data === 'object') {
                        // Map the extracted data to form fields
                        Object.keys(financialData).forEach(key => {
                            if (data[key]) financialData[key] = data[key];
                        });
                        alert('Data extracted! Please verify the fields below.');
                    }
                } catch (error) {
                    console.error("Upload Error:", error);
                    alert('Extraction failed. Please enter data manually below.');
                } finally {
                    loading = false;
                    render();
                }
            };
            reader.readAsDataURL(file);
        }

        function calculateRatios(data) {
            const revenue = parseFloat(data.revenue) || 0;
            const cogs = parseFloat(data.cogs) || 0;
            const opex = parseFloat(data.opex) || 0;
            const assets = parseFloat(data.assets) || 0;
            const liabilities = parseFloat(data.liabilities) || 0;
            const equity = parseFloat(data.equity) || 0;

            const netIncome = revenue - cogs - opex;
            const grossMargin = revenue > 0 ? ((revenue - cogs) / revenue * 100) : 0;
            const netMargin = revenue > 0 ? (netIncome / revenue * 100) : 0;
            const currentRatio = liabilities > 0 ? (assets / liabilities) : 0;
            const debtToEquity = equity > 0 ? (liabilities / equity) : 0;
            const roe = equity > 0 ? (netIncome / equity * 100) : 0;

            return {
                netIncome,
                grossMargin,
                netMargin,
                currentRatio,
                debtToEquity,
                roe
            };
        }

        async function analyzeFinancials() {
            loading = true;
            render();
            const ratios = calculateRatios(financialData);

            try {
                const response = await fetch('https://billbotprocessing.app.n8n.cloud/webhook/varia-analysis', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ period: financialData.period, financialData, ratios, includeSummary })
                });

                const result = await response.json();
                analysis = { 
                    ratios, 
                    insights: cleanFormatting(result.insights), 
                    summary: cleanFormatting(result.summary), 
                    data: financialData 
                };
                step = 'results';
            } catch (error) {
                alert('Analysis failed. Check your n8n workflow.');
            } finally {
                loading = false;
                render();
            }
        }
